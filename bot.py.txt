def generate_cp(rarity: str, is_shiny: bool, is_shadow: bool = False) -> int:
    min_cp, max_cp = CP_RANGES.get(rarity, (100, 500))
    cp = random.randint(min_cp, max_cp)
    if is_shiny: cp = int(cp * 1.15)
    if is_shadow: cp = int(cp * 1.20)
    return cp

def get_user_data(user_id: int, username: str = ""):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT * FROM users WHERE user_id = ?", (user_id,))
    u = cur.fetchone()
    now = time.time()
    
    if not u:
        cur.execute("""
            INSERT INTO users (user_id, username, coins, candies, energy, last_energy_calc)
            VALUES (?, ?, 500, 10, 100, ?)
        """, (user_id, username, now))
        # Стартовый покемон (Пикачу)
        cur.execute("""
            INSERT INTO inventory (user_id, pokemon_id, pokemon_name, gen, rarity, is_shiny, is_shadow, is_purified, cp, in_squad)
            VALUES (?, 25, 'Пикачу', 1, 'Rare', 0, 0, 0, 480, 1)
        """, (user_id,))
        conn.commit()
        cur.close()
        conn.close()
        return get_user_data(user_id, username)

    col_names = [desc[0] for desc in cur.description]
    user_dict = dict(zip(col_names, u))

    # Корректный серверный расчёт энергии
    last_calc = user_dict.get("last_energy_calc", now)
    current_energy = user_dict.get("energy", 100)
    if current_energy < MAX_ENERGY:
        recovered = int((now - last_calc) / ENERGY_RECOVERY_SECONDS)
        if recovered > 0:
            new_energy = min(MAX_ENERGY, current_energy + recovered)
            cur.execute("UPDATE users SET energy = ?, last_energy_calc = ? WHERE user_id = ?", (new_energy, now, user_id))
            conn.commit()
            user_dict["energy"] = new_energy
            user_dict["last_energy_calc"] = now

    cur.close()
    conn.close()
    return user_dict

def update_user(user_id: int, **kwargs):
    conn = get_db()
    cur = conn.cursor()
    for k, v in kwargs.items():
        cur.execute(f"UPDATE users SET {k} = ? WHERE user_id = ?", (v, user_id))
    conn.commit()
    cur.close()
    conn.close()

def change_coins(user_id: int, delta: int):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("UPDATE users SET coins = MAX(0, COALESCE(coins, 0) + ?) WHERE user_id = ?", (delta, user_id))
    conn.commit()
    cur.close()
    conn.close()

def change_candies(user_id: int, delta: int):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("UPDATE users SET candies = MAX(0, COALESCE(candies, 0) + ?) WHERE user_id = ?", (delta, user_id))
    conn.commit()
    cur.close()
    conn.close()

def track_achievement(user_id: int, achieve_id: int, add_progress: int = 1):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("""
        INSERT INTO achievements (user_id, achieve_id, progress, claimed)
        VALUES (?, ?, ?, 0)
        ON CONFLICT(user_id, achieve_id) DO UPDATE SET progress = progress + ?
    """, (user_id, achieve_id, add_progress, add_progress))
    conn.commit()
    cur.close()
    conn.close()

@web.middleware
async def cors_middleware(request, handler):
    if request.method == "OPTIONS":
        response = web.Response(status=200)
    else:
        response = await handler(request)
    response.headers["Access-Control-Allow-Origin"] = "*"
    response.headers["Access-Control-Allow-Methods"] = "GET, POST, OPTIONS"
    response.headers["Access-Control-Allow-Headers"] = "Content-Type, Authorization"
    return response

            cp = random.randint(1200, 2400)
            return web.json_response({
                "status": "rocket_ambush",
                "enemy": {"id": pid, "name": f"Теневой {name}", "gen": gen, "rarity": "Rare", "is_shadow": True, "cp": cp, "type": POKEMON_TYPES.get(pid, "⚪️")}
            })

        is_ultra = ball_type == "ultra" and u["ultraballs"] > 0
        if is_ultra:
            update_user(uid, ultraballs=u["ultraballs"] - 1)

        roll = random.randint(1, 100)
        rarity = ("Rare" if roll <= 50 else "Epic" if roll <= 90 else "Legendary") if is_ultra else ("Common" if roll <= 60 else "Uncommon" if roll <= 85 else "Rare")
        
        candidates = POKEMON_DB.get(rarity, POKEMON_DB["Common"])
        pid, name, gen = random.choice(candidates)
        
        shiny = 1 if random.random() < 0.04 else 0
        cp = generate_cp(rarity, bool(shiny))
        
        save_pokemon(uid, pid, name, gen, rarity, shiny, 0, 0, cp)
        change_coins(uid, 20)
        track_achievement(uid, 1, 1)
        track_achievement(uid, 2, 1)
        if shiny: track_achievement(uid, 6, 1)

        return web.json_response({
            "status": "success",
            "pokemon": {
                "id": pid, "name": name, "gen": gen, "rarity": rarity,
                "is_shiny": bool(shiny), "cp": cp, "type": POKEMON_TYPES.get(pid, "⚪️ Обычный")
            }
        })
    except Exception as e:
        return web.json_response({"status": "error", "message": str(e)}, status=500)

async def api_evolve_handler(request):
    try:
        data = await request.json()
        uid = int(data.get("user_id", 0))
        inv_id = int(data.get("inv_id", 0))

        conn = get_db()
        cur = conn.cursor()
        cur.execute("SELECT pokemon_id, is_shiny, is_shadow FROM inventory WHERE id = ? AND user_id = ?", (inv_id, uid))
        row = cur.fetchone()
        if not row:
            cur.close()
            conn.close()
            return web.json_response({"status": "error", "message": "Покемон не найден!"})

        old_pid, shiny, shadow = row[0], row[1], row[2]
        if old_pid not in EVOLUTIONS:
            cur.close()
            conn.close()
            return web.json_response({"status": "error", "message": "Этот покемон достиг финальной формы!"})

        next_pid = EVOLUTIONS[old_pid]
        next_info = ALL_POKEMON_MAP.get(next_pid, {"name": "Эволюция", "rarity": "Epic", "gen": 1})

        u = get_user_data(uid)
        if u["candies"] < 3 or u["coins"] < 150:
            cur.close()
            conn.close()
            return web.json_response({"status": "error", "message": "Нужно: 3 🍬 и 150 🪙!"})

        new_cp = generate_cp(next_info["rarity"], bool(shiny), bool(shadow)) + 400
        cur.execute("""
            UPDATE inventory 
            SET pokemon_id = ?, pokemon_name = ?, gen = ?, rarity = ?, cp = ?
            WHERE id = ? AND user_id = ?
        """, (next_pid, next_info["name"], next_info["gen"], next_info["rarity"], new_cp, inv_id, uid))
        conn.commit()
        cur.close()
        conn.close()

        change_candies(uid, -3)
        change_coins(uid, -150)
        track_achievement(uid, 11, 1)

        return web.json_response({
            "status": "success",
            "pokemon": {"id": next_pid, "name": next_info["name"], "cp": new_cp, "rarity": next_info["rarity"]}
        })
    except Exception as e:
        return web.json_response({"status": "error", "message": str(e)}, status=500)

async def api_claim_achievement_handler(request):
    try:
        data = await request.json()
        uid = int(data.get("user_id", 0))
        aid = int(data.get("achieve_id", 0))

        conn = get_db()
        cur = conn.cursor()
        cur.execute("SELECT progress, claimed FROM achievements WHERE user_id = ? AND achieve_id = ?", (uid, aid))
        row = cur.fetchone()
        if not row or row[1] == 1:
            cur.close()
            conn.close()
            return web.json_response({"status": "error", "message": "Награда уже получена!"})

        cur.execute("UPDATE achievements SET claimed = 1 WHERE user_id = ? AND achieve_id = ?", (uid, aid))
        conn.commit()
        cur.close()
        conn.close()

        change_coins(uid, 150)
        change_candies(uid, 3)

        return web.json_response({"status": "success", "message": "+150 🪙 и +3 🍬 начислены!"})
    except Exception as e:
        return web.json_response({"status": "error", "message": str(e)}, status=500)

async def api_admin_give_handler(request):
    try:
        data = await request.json()
        admin_id = int(data.get("admin_id", 0))
        if not check_admin(admin_id):
            return web.json_response({"status": "error", "message": "Доступ запрещен!"}, status=403)

        target_id = int(data.get("target_id", admin_id))
        item_type = data.get("type", "")

        if item_type == "coins":
            amt = int(data.get("amount", 1000))
            change_coins(target_id, amt)
        elif item_type == "pokemon":
            pid = int(data.get("pokemon_id", 25))
            shiny = 1 if data.get("shiny") else 0
            pinfo = ALL_POKEMON_MAP.get(pid, {"name": "Пикачу", "rarity": "Rare", "gen": 1})
            cp = int(data.get("cp", 3500))
            save_pokemon(target_id, pid, pinfo["name"], pinfo["gen"], pinfo["rarity"], shiny, 0, 0, cp)
        elif item_type == "egg":
            update_user(target_id, egg_type="Legendary", egg_hatch_time=time.time() + 10)

        return web.json_response({"status": "success", "message": "Успешно выдано!"})
    except Exception as e:
        return web.json_response({"status": "error", "message": str(e)}, status=500)
