def get_pokemon_image_url(pokemon_id: int, is_shiny: bool) -> str:
    shiny_path = "shiny/" if is_shiny else ""
    return f"https://raw.githubusercontent.com/PokeAPI/sprites/master/sprites/pokemon/other/official-artwork/{shiny_path}{pokemon_id}.png"

def generate_cp(rarity: str, is_shiny: bool, is_shadow: bool = False, is_purified: bool = False) -> int:
    min_cp, max_cp = CP_RANGES[rarity]
    cp = random.randint(min_cp, max_cp)
    if is_shiny: cp = int(cp * 1.15)
    if is_shadow: cp = int(cp * 1.20)
    if is_purified: cp = int(cp * 1.10)
    return cp

def get_user_data(user_id: int, username: str = ""):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT * FROM users WHERE user_id = %s", (user_id,))
    u = cur.fetchone()
    if not u:
        cur.execute("INSERT INTO users (user_id, username, last_payout) VALUES (%s, %s, %s) ON CONFLICT DO NOTHING", (user_id, username, time.time()))
        cur.execute("""
            INSERT INTO users (user_id, username, coins, candies, potions, quiz_attempts)
            VALUES (%s, %s, 50, 2, 2, 3)
        """, (user_id, username))
        conn.commit()
        return get_user_data(user_id, username)
    
    col_names = [desc[0] for desc in cur.description]
    user_dict = dict(zip(col_names, u))
    if username and user_dict.get("username") != username:
        cur.execute("UPDATE users SET username = %s WHERE user_id = %s", (username, user_id))
        conn.commit()
        user_dict["username"] = username
    cur.close()
    conn.close()
    return user_dict

def update_user(user_id: int, **kwargs):
    conn = get_db()
    cur = conn.cursor()
    for k, v in kwargs.items():
        cur.execute(f"UPDATE users SET {k} = %s WHERE user_id = %s", (v, user_id))
    conn.commit()
    cur.close()
    conn.close()

def change_coins(user_id: int, delta: int):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("UPDATE users SET coins = GREATEST(0, COALESCE(coins, 0) + %s) WHERE user_id = %s", (delta, user_id))
    conn.commit()
    cur.close()
    conn.close()

def change_candies(user_id: int, delta: int):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("UPDATE users SET candies = GREATEST(0, COALESCE(candies, 0) + %s) WHERE user_id = %s", (delta, user_id))
    conn.commit()
    cur.close()
    conn.close()

def save_pokemon(user_id: int, pid: int, name: str, rarity: str, shiny: int, shadow: int, purified: int, cp: int) -> bool:
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT id FROM inventory WHERE user_id = %s AND pokemon_name = %s AND is_shiny = %s AND is_shadow = %s", (user_id, name, shiny, shadow))
    if cur.fetchone():
        cur.close()
        conn.close()
        return False
    cur.execute("""
        INSERT INTO inventory (user_id, pokemon_id, pokemon_name, rarity, is_shiny, is_shadow, is_purified, cp)
        VALUES (%s, %s, %s, %s, %s, %s, %s, %s)
    """, (user_id, pid, name, rarity, shiny, shadow, purified, cp))
    conn.commit()
    cur.close()
    conn.close()
    return True

def get_inventory(user_id: int):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT id, pokemon_id, pokemon_name, rarity, is_shiny, is_shadow, is_purified, cp FROM inventory WHERE user_id = %s ORDER BY cp DESC", (user_id,))
    rows = cur.fetchall()
    cur.close()
    conn.close()
    return rows

ACTIVE_BATTLES = {}

def build_hp_bar(current_hp: int, max_hp: int) -> str:
    pct = max(0, min(1.0, current_hp / max_hp))
    filled = int(round(pct * 8))
    empty = 8 - filled
    return "🟩" * filled + "⬜️" * empty

    pid, name, rarity, is_shiny, is_shadow, is_purified, cp = p[1], p[2], p[3], p[4], p[5], p[6], p[7]

        f"🏷 **{name}** #{pid:03d}{status_tag}\n"

        f"🧬 Стихия: {ptype}\n"
        f"⚔️ Мощь: **{cp} CP**"
    )
    
    builder = InlineKeyboardBuilder()
    prev_idx = len(pokes) - 1 if idx == 0 else idx - 1
    next_idx = 0 if idx == len(pokes) - 1 else idx + 1
    
    builder.row(
        InlineKeyboardButton(text="⬅️ Назад", callback_data=f"pdx_{user_id}_{prev_idx}"),
        InlineKeyboardButton(text=f"{idx+1} / {len(pokes)}", callback_data="none"),

    )
    
    img_url = get_pokemon_image_url(pid, bool(is_shiny))
    
    if edit:
        try:

# ----------------- БОЕВАЯ СИСТЕМА (ОБРАБОТКА ХОДОВ) -----------------
@dp.callback_query(F.data.startswith("b_act_"))
async def battle_action_handler(callback: CallbackQuery):
    _, _, b_id, action = callback.data.split("_")
    b = ACTIVE_BATTLES.get(b_id)
    if not b:
        await callback.answer("Бой уже завершен!", show_alert=True)
        return

    # Проверка, чей ход (в PvP)
    attacker = b["p1"] if callback.from_user.id == b["p1"]["id"] else b["p2"]
    defender = b["p2"] if callback.from_user.id == b["p1"]["id"] else b["p1"]
    
    if callback.from_user.id != attacker["id"]:
        await callback.answer("Сейчас не ваш ход!", show_alert=True)
        return

    log_event = ""
    # Действие игрока
    if action == "atk":
        dmg = random.randint(18, 28)
        if defender["shield"]: dmg = int(dmg * 0.25)
        defender["hp"] = max(0, defender["hp"] - dmg)
        attacker["sp"] = min(3, attacker["sp"] + 1)
        log_event = f"⚔️ {attacker['name']} нанес быстрый удар: -{dmg} HP!"
    elif action == "spc":
        if attacker["sp"] < 2:
            await callback.answer("Нужно минимум 2 SP для супер-удара!", show_alert=True)
            return
        attacker["sp"] -= 2
        mult = 1.5 if defender["type"] in TYPE_ADVANTAGE.get(attacker["type"], []) else 1.0
        dmg = int(random.randint(40, 58) * mult)
        if defender["shield"]: dmg = int(dmg * 0.25)
        defender["hp"] = max(0, defender["hp"] - dmg)
        bonus_str = " (Стихийный бонус!)" if mult > 1.0 else ""
        log_event = f"💥 {attacker['name']} обрушил супер-удар{bonus_str}: -{dmg} HP!"
    elif action == "def":
        attacker["shield"] = True
        log_event = f"🛡 {attacker['name']} встал в глухую защиту!"
    elif action == "pot":
        if attacker["potions"] <= 0:
            await callback.answer("Зелья закончились!", show_alert=True)
            return
        attacker["potions"] -= 1
        heal = int(attacker["max_hp"] * 0.40)
        attacker["hp"] = min(attacker["max_hp"], attacker["hp"] + heal)
        log_event = f"🧪 {attacker['name']} выпил зелье и восстановил +{heal} HP!"

    # Сброс щита цели на следующий раунд
    defender["shield"] = False

    # Проверка победы игрока
    if defender["hp"] <= 0:
        await finish_battle(callback, b, winner=attacker, loser=defender)
        return

    # Ответный ход Бота / Защитника башни
    if b["p2"].get("id") == 0 or b["type"] in ["rocket", "tower"]:
        bot = b["p2"]
        user_p = b["p1"]
        b_act = "spc" if bot["sp"] >= 2 and random.random() < 0.6 else "atk"
        if b_act == "spc":
            bot["sp"] -= 2
            b_dmg = int(random.randint(35, 50))
            if user_p["shield"]: b_dmg = int(b_dmg * 0.25)
            user_p["hp"] = max(0, user_p["hp"] - b_dmg)
            log_event += f"\n🤖 Враг ответил супер-ударом: -{b_dmg} HP!"
        else:
            b_dmg = random.randint(15, 25)
            if user_p["shield"]: b_dmg = int(b_dmg * 0.25)
            user_p["hp"] = max(0, user_p["hp"] - b_dmg)
            bot["sp"] = min(3, bot["sp"] + 1)
            log_event += f"\n🤖 Враг атаковал в ответ: -{b_dmg} HP!"
        
        user_p["shield"] = False
        if user_p["hp"] <= 0:
            await finish_battle(callback, b, winner=bot, loser=user_p)
            return

    b["round"] += 1
    b["log"] = log_event
    new_text, new_markup = render_battle_card(b)
    try:
        await callback.message.edit_text(new_text, reply_markup=new_markup, parse_mode="Markdown")
    except Exception:
        pass
    await callback.answer()

async def finish_battle(callback: CallbackQuery, b: dict, winner: dict, loser: dict):
    b_id = b["id"]
    del ACTIVE_BATTLES[b_id]
    
    if b["type"] == "rocket":
        if winner["id"] == b["p1"]["id"]:
            meta = b["metadata"]
            save_pokemon(winner["id"], meta["reward_pid"], meta["reward_name"], "Rare", 0, 1, 0, meta["reward_cp"])
            change_coins(winner["id"], 35)
            await callback.message.edit_text(
                f"🎉 **ПОБЕДА НАД КОМАНДОЙ R!**\n\n"
                f"Вы спасли покемона: **☠️ Теневой {meta['reward_name']} ({meta['reward_cp']} CP)**!\n"
                f"💰 Награда: **+35 🪙 монет**!\n"
                f"💡 *Очистите его в Питомнике для бонуса к статам.*",
                parse_mode="Markdown"
            )
        else:
            await callback.message.edit_text("💀 **ПОРАЖЕНИЕ!** Бандиты Команды R оказались сильнее.", parse_mode="Markdown")
    elif b["type"] == "tower":
        if winner["id"] == b["p1"]["id"]:
            tid = b["metadata"]["tower_id"]
            conn = get_db()
            cur = conn.cursor()
            cur.execute("""
                UPDATE towers 
                SET champ_id = %s, champ_name = %s, p_name = %s, p_id = %s, p_cp = %s, last_payout = %s
                WHERE tower_id = %s
            """, (winner["id"], winner["name"], winner["poke_name"], winner["poke_id"], winner["cp"], time.time(), tid))
            conn.commit()
            cur.close()
            conn.close()
            await callback.message.edit_text(
                f"👑 **ВЫ ЗАХВАТИЛИ БАШНЮ!**\n\n"
                f"Ваш **{winner['poke_name']}** занял трон.\n"
                f"Теперь вы получаете пассивный доход каждый час!",
                parse_mode="Markdown"
            )
        else:
            await callback.message.edit_text("💀 **ШТУРМ ПРОВАЛЕН!** Защитник башни отбил вашу атаку.", parse_mode="Markdown")
    elif b["type"] == "duel":
        change_coins(winner["id"], 25)
        change_coins(loser["id"], -25)
        await callback.message.edit_text(
            f"🏆 **ДУЭЛЬ ЗАВЕРШЕНА!**\n\n"
            f"Победитель: **{winner['name']}** (+25 🪙)\n"
            f"Проигравший: **{loser['name']}** (-25 🪙)",
            parse_mode="Markdown"
        )

        return

    if not admin_mode:
        update_user(message.from_user.id, quiz_attempts=attempts - 1)

    pid = random.randint(1, 151)
    c_name = ALL_POKEMON_MAP[pid]["name"]
    c_type = POKEMON_TYPES.get(pid, "⚪️ Обычный")

    all_t = list(set(POKEMON_TYPES.values()))
    wrongs = random.sample([t for t in all_t if t != c_type], 3)
    opts = wrongs + [c_type]
    random.shuffle(opts)

    b = InlineKeyboardBuilder()
    for opt in opts:
        b.row(InlineKeyboardButton(text=opt, callback_data=f"qz_{'1' if opt==c_type else '0'}"))

    await message.answer(f"🧩 **ВИКТОРИНА (Осталось: {attempts}/3):**\nК какой стихии относится **{c_name}**?", reply_markup=b.as_markup(), parse_mode="Markdown")

@dp.callback_query(F.data.startswith("buy_lot_"))
async def buy_market_lot(callback: CallbackQuery):
    lid = int(callback.data.split("_")[2])
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT lot_id, seller_id, pokemon_id, pokemon_name, rarity, is_shiny, is_shadow, cp, price FROM market WHERE lot_id = %s", (lid,))
    lot = cur.fetchone()
    if not lot:
        cur.close()
        conn.close()
        await callback.answer("Лот уже продан!", show_alert=True)
        return

    u = get_user_data(callback.from_user.id, callback.from_user.username or "")
    if u["coins"] < lot[8]:
        cur.close()
        conn.close()
        await callback.answer("Недостаточно монет!", show_alert=True)
        return

    # Перевод
    change_coins(callback.from_user.id, -lot[8])
    change_coins(lot[1], lot[8])
    save_pokemon(callback.from_user.id, lot[2], lot[3], lot[4], lot[5], lot[6], 0, lot[7])
    cur.execute("DELETE FROM market WHERE lot_id = %s", (lid,))
    conn.commit()
    cur.close()
    conn.close()

    await callback.message.edit_text(f"🎉 **Вы успешно купили {lot[3]} за {lot[8]} 🪙!**")
    await callback.answer()
