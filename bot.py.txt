def generate_cp(rarity: str, is_shiny: bool) -> int:
    min_cp, max_cp = CP_RANGES[rarity]
    cp = random.randint(min_cp, max_cp)
    if is_shiny:
        cp = int(cp * 1.15)
    return cp

def init_db():
    conn = sqlite3.connect("pokemon_bot.db")
    cur = conn.cursor()
    cur.execute("""
        CREATE TABLE IF NOT EXISTS users (
            user_id INTEGER PRIMARY KEY,
            username TEXT,
            coins INTEGER DEFAULT 0,
            candies INTEGER DEFAULT 0,
            last_hunt REAL DEFAULT 0,
            lures INTEGER DEFAULT 0,
            ultraballs INTEGER DEFAULT 0,
            last_daily REAL DEFAULT 0,
            egg_rarity TEXT DEFAULT NULL,
            egg_hatch_time REAL DEFAULT 0,
            referred_by INTEGER DEFAULT 0,
            vip_until REAL DEFAULT 0,
            expedition_tier INTEGER DEFAULT 0,
            expedition_end REAL DEFAULT 0,
            quiz_attempts INTEGER DEFAULT 3,
            quiz_reset_time REAL DEFAULT 0,
            gym_stage INTEGER DEFAULT 0
        )
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS inventory (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER,
            pokemon_id INTEGER DEFAULT 0,
            pokemon_name TEXT,
            rarity TEXT,
            is_shiny INTEGER DEFAULT 0,
            cp INTEGER DEFAULT 100,
            UNIQUE(user_id, pokemon_name, is_shiny)
        )
    """)
    conn.commit()

    # Миграция колонок для существующих БД
    columns = [
        ("username", "TEXT"), ("candies", "INTEGER DEFAULT 0"),
        ("lures", "INTEGER DEFAULT 0"), ("ultraballs", "INTEGER DEFAULT 0"),
        ("last_daily", "REAL DEFAULT 0"), ("egg_rarity", "TEXT DEFAULT NULL"),
        ("egg_hatch_time", "REAL DEFAULT 0"), ("referred_by", "INTEGER DEFAULT 0"),
        ("vip_until", "REAL DEFAULT 0"), ("expedition_tier", "INTEGER DEFAULT 0"),
        ("expedition_end", "REAL DEFAULT 0"), ("quiz_attempts", "INTEGER DEFAULT 3"),
        ("quiz_reset_time", "REAL DEFAULT 0"), ("gym_stage", "INTEGER DEFAULT 0")
    ]
    for col, defn in columns:
        try:
            cur.execute(f"ALTER TABLE users ADD COLUMN {col} {defn}")
            conn.commit()
        except sqlite3.OperationalError:
            pass

    for col, defn in [("pokemon_id", "INTEGER DEFAULT 0"), ("cp", "INTEGER DEFAULT 100")]:
        try:
            cur.execute(f"ALTER TABLE inventory ADD COLUMN {col} {defn}")
            conn.commit()
        except sqlite3.OperationalError:
            pass
    conn.close()

def get_user_data(user_id: int, username: str):
    conn = sqlite3.connect("pokemon_bot.db")
    cur = conn.cursor()
    cur.execute("""
        SELECT user_id, coins, last_hunt, lures, ultraballs, last_daily, username, 
               candies, egg_rarity, egg_hatch_time, vip_until, expedition_tier, 
               expedition_end, quiz_attempts, quiz_reset_time, gym_stage, referred_by 
        FROM users WHERE user_id = ?
    """, (user_id,))
    user = cur.fetchone()
    if not user:
        cur.execute("""
            INSERT INTO users (user_id, username, coins, candies, last_hunt, lures, ultraballs, last_daily, quiz_attempts, quiz_reset_time, gym_stage) 
            VALUES (?, ?, 0, 0, 0, 0, 0, 0, 3, 0, 0)
        """, (user_id, username))
        conn.commit()
        return get_user_data(user_id, username)
    else:
        if username and user[6] != username:
            cur.execute("UPDATE users SET username = ? WHERE user_id = ?", (username, user_id))
            conn.commit()
    conn.close()
    return {
        "user_id": user[0], "coins": user[1], "last_hunt": user[2],
        "lures": user[3], "ultraballs": user[4], "last_daily": user[5],
        "username": user[6], "candies": user[7], "egg_rarity": user[8],
        "egg_hatch_time": user[9], "vip_until": user[10], "expedition_tier": user[11],
        "expedition_end": user[12], "quiz_attempts": user[13], "quiz_reset_time": user[14],
        "gym_stage": user[15] if len(user) > 15 else 0,
        "referred_by": user[16] if len(user) > 16 else 0
    }

def update_user_field(user_id: int, **kwargs):
    conn = sqlite3.connect("pokemon_bot.db")
    cur = conn.cursor()
    for key, value in kwargs.items():
        cur.execute(f"UPDATE users SET {key} = ? WHERE user_id = ?", (value, user_id))
    conn.commit()
    conn.close()

    conn.commit()
    conn.close()

    conn.commit()
    conn.close()

def save_pokemon(user_id: int, pokemon_id: int, name: str, rarity: str, is_shiny: int, cp: int) -> bool:
    conn = sqlite3.connect("pokemon_bot.db")
    cur = conn.cursor()
    cur.execute("SELECT id FROM inventory WHERE user_id = ? AND pokemon_name = ? AND is_shiny = ?", (user_id, name, is_shiny))
    exists = cur.fetchone()
    if exists:
        conn.close()
        return False
    cur.execute(
        "INSERT INTO inventory (user_id, pokemon_id, pokemon_name, rarity, is_shiny, cp) VALUES (?, ?, ?, ?, ?, ?)",
        (user_id, pokemon_id, name, rarity, is_shiny, cp)
    )
    conn.commit()
    conn.close()
    return True

def get_pokedex(user_id: int):
    conn = sqlite3.connect("pokemon_bot.db")
    cur = conn.cursor()
    cur.execute("SELECT pokemon_id, pokemon_name, rarity, is_shiny, cp FROM inventory WHERE user_id = ? ORDER BY cp DESC", (user_id,))
    rows = cur.fetchall()
    conn.close()
    return rows

def get_top_players():
    conn = sqlite3.connect("pokemon_bot.db")
    cur = conn.cursor()
    cur.execute("""
        SELECT u.username, COUNT(i.id) as count, COALESCE(SUM(i.cp), 0) as total_cp, u.gym_stage
        FROM users u
        LEFT JOIN inventory i ON u.user_id = i.user_id
        GROUP BY u.user_id
        ORDER BY count DESC, total_cp DESC
        LIMIT 10
    """)
    rows = cur.fetchall()
    conn.close()
    return rows

def replace_pokemon_for_evolve(user_id: int, old_id: int, new_id: int, new_name: str, new_rarity: str, new_cp: int):
    conn = sqlite3.connect("pokemon_bot.db")
    cur = conn.cursor()
    cur.execute("DELETE FROM inventory WHERE user_id = ? AND pokemon_id = ?", (user_id, old_id))
    cur.execute(
        "INSERT INTO inventory (user_id, pokemon_id, pokemon_name, rarity, is_shiny, cp) VALUES (?, ?, ?, ?, 0, ?)",
        (user_id, new_id, new_name, new_rarity, new_cp)
    )

@dp.callback_query(F.data == "fight_gym_boss")
async def fight_gym_boss(callback: CallbackQuery):
    user = get_user_data(callback.from_user.id, callback.from_user.username or "")
    stage = user["gym_stage"]

    if stage >= len(GYM_LEADERS):
        await callback.answer("Вы уже прошли все стадионы!", show_alert=True)
        return

    pokes = get_pokedex(user["user_id"])
    if not pokes:
        await callback.answer("❌ У вас нет покемонов для битвы!", show_alert=True)
        return

    current_gym = GYM_LEADERS[stage]
    boss_type = POKEMON_TYPES.get(current_gym["pid"], "⚪️ Обычный")
    boss_cp = current_gym["cp"]

    p1 = pokes[0] # Сильнейший покемон игрока
    p1_type = POKEMON_TYPES.get(p1[0], "⚪️ Обычный")

    # Расчет бонусов стихий
    p1_mult = 1.35 if boss_type in TYPE_ADVANTAGE.get(p1_type, []) else 1.0
    boss_mult = 1.35 if p1_type in TYPE_ADVANTAGE.get(boss_type, []) else 1.0

    final_p1_cp = int(p1[4] * p1_mult)
    final_boss_cp = int(boss_cp * boss_mult)

    p1_bonus_tag = " *(Эффективно! x1.35)*" if p1_mult > 1.0 else ""
    boss_bonus_tag = " *(Эффективно! x1.35)*" if boss_mult > 1.0 else ""

    if final_p1_cp >= final_boss_cp:
        # Победа!
        new_stage = stage + 1
        change_user_coins(user["user_id"], current_gym["reward_c"])
        change_user_candies(user["user_id"], current_gym["reward_k"])
        update_user_field(user["user_id"], gym_stage=new_stage)

        extra_reward = ""
        if new_stage == 8:
            update_user_field(user["user_id"], egg_rarity="Epic", egg_hatch_time=time.time() + 6*3600)
            extra_reward = "\n🎁 **Вам подарено Эпическое Яйцо в Инкубатор!**"

        await callback.message.delete()
        res_text = (
            f"🎉 **ВЕЛИКАЯ ПОБЕДА НАД {current_gym['leader'].upper()}!**\n\n"
            f"🔴 Ваш боец: **{p1[1]}** ({p1_type}) — {final_p1_cp} CP{p1_bonus_tag}\n"
            f"🔵 Босс: **{current_gym['poke']}** ({boss_type}) — {final_boss_cp} CP{boss_bonus_tag}\n\n"
            f"🏆 **Вы заслужили:** {current_gym['badge']}!\n"
            f"💰 Награды: **+{current_gym['reward_c']} 🪙 монет** и **+{current_gym['reward_k']} 🍬 конфет**!{extra_reward}"
        )
        await callback.message.answer(res_text, parse_mode="Markdown")
    else:
        diff = final_boss_cp - final_p1_cp
        await callback.message.delete()
        res_text = (
            f"💀 **ПОРАЖЕНИЕ НА СТАДИОНЕ!**\n\n"
            f"🔴 Ваш боец: **{p1[1]}** ({p1_type}) — {final_p1_cp} CP{p1_bonus_tag}\n"
            f"🔵 Босс: **{current_gym['poke']}** ({boss_type}) — {final_boss_cp} CP{boss_bonus_tag}\n\n"
            f"Вам не хватило около **{diff} CP**. Эволюционируйте покемонов или подберите бойца с выгодной стихией!"
        )
        await callback.message.answer(res_text, parse_mode="Markdown")

    await callback.answer()

@dp.message(F.text == "🌿 Искать в траве")
@dp.message(Command("hunt"))
async def hunt_pokemon(message: types.Message):
    user_id = message.from_user.id
    user_uname = message.from_user.username or ""
    user = get_user_data(user_id, user_uname)
    admin_mode = is_admin(user_uname)
    now = time.time()
    
    cooldown = COOLDOWN_HUNT_VIP if now < user["vip_until"] else COOLDOWN_HUNT_DEFAULT
    
    if not admin_mode and (now - user["last_hunt"]) < cooldown:
        rem = int(cooldown - (now - user["last_hunt"]))
        await message.answer(f"⏳ Вы устали. Отдых: **{rem//3600} ч. {(rem%3600)//60} мин.**\n💡 Используйте Приманку в Магазине!", parse_mode="Markdown")
        return

    # Засада Команды R (10%)
    if random.random() < 0.10:
        if not admin_mode:
            update_user_field(user_id, last_hunt=now)
        user_pokes = get_pokedex(user_id)
        best_cp = user_pokes[0][4] if user_pokes else 150
        rocket_cp = random.randint(300, 1500)
        
        if best_cp >= rocket_cp:
            change_user_coins(user_id, 25)
            change_user_candies(user_id, 2)
            await message.answer(
                f"🏴‍☠️ **НАПАДЕНИЕ КОМАНДЫ R!**\n\n"
                f"Их боец ({rocket_cp} CP) атаковал вас, но ваш сильнейший боец победил!\n"
                f"🎉 Трофеи: **+25 🪙 монет и +2 🍬 конфеты**!",
                parse_mode="Markdown"
            )
        else:
            await message.answer(f"🏴‍☠️ **КОМАНДА R!** Их боец ({rocket_cp} CP) был сильнее, вам пришлось убежать!", parse_mode="Markdown")
        return

    # Находка Яйца (12%)
    if not user["egg_rarity"] and random.random() < 0.12:
        egg_type = random.choice(["Rare", "Epic"])
        hatch_time = now + (2 * 3600 if egg_type == "Rare" else 6 * 3600)
        update_user_field(user_id, egg_rarity=egg_type, egg_hatch_time=hatch_time)
        await message.answer(f"🥚 **Вы нашли Таинственное Яйцо ({RARITY_NAMES[egg_type]})!**\nОно помещено в **🥚 Инкубатор**.", parse_mode="Markdown")

    # Ловля покемона
    used_ultraball = user["ultraballs"] > 0
    if used_ultraball:
        update_user_field(user_id, ultraballs=user["ultraballs"] - 1)

    roll = random.randint(1, 100)
    if not used_ultraball and roll <= 20:
        if not admin_mode:
            update_user_field(user_id, last_hunt=now)
        await message.answer("💨 Покемон испугался и скрылся в траве!", parse_mode="Markdown")
        return

    if used_ultraball:
        rarity = "Common" if roll <= 45 else "Rare" if roll <= 75 else "Epic" if roll <= 92 else "Legendary"
    else:
        rarity = "Common" if roll <= 70 else "Rare" if roll <= 90 else "Epic" if roll <= 98 else "Legendary"

    pid, name = random.choice(POKEMON_DB[rarity])
    is_shiny = 1 if random.random() < 0.05 else 0
    shiny_p = "✨ ШАЙНИ " if is_shiny else ""
    cp = generate_cp(rarity, bool(is_shiny))
    is_new = save_pokemon(user_id, pid, name, rarity, is_shiny, cp)
    
    if not admin_mode:
        update_user_field(user_id, last_hunt=now)

    ball_tag = "\n🔘 *Активирован Ультрабол (100% поимка)!*" if used_ultraball else ""
    image_url = get_pokemon_image_url(pid, bool(is_shiny))

    if is_new:
        caption = (
            f"🎉 **Вы поймали покемона:**{ball_tag}\n\n"
            f"🏷 **№{pid:03d} {shiny_p}{name}**\n"
            f"🌟 Редкость: {RARITY_NAMES[rarity]} | 🧬 {POKEMON_TYPES.get(pid, '⚪️')}\n"
            f"⚔️ Мощь: **{cp} CP**\n\n"
            f"📖 *Записан в Покедекс!*"
        )
    else:
        reward_c = COIN_REWARDS[rarity]
        reward_k = 1 if rarity in ["Common", "Rare"] else 2
        change_user_coins(user_id, reward_c)
        change_user_candies(user_id, reward_k)
        caption = (
            f"📦 Повторка: **№{pid:03d} {shiny_p}{name}** ({cp} CP).{ball_tag}\n\n"
            f"Вы отпустили его:\n• **+{reward_c} 🪙 монет**\n• **+{reward_k} 🍬 конфеты**"
        )

    try:
        await message.answer_photo(photo=image_url, caption=caption, parse_mode="Markdown")

@dp.callback_query(F.data.startswith("pvp_acc_"))
async def process_pvp_acc(callback: CallbackQuery):
    _, _, c_id_str, t_id_str = callback.data.split("_")
    c_id, t_id = int(c_id_str), int(t_id_str)

    if callback.from_user.id != t_id:
        await callback.answer("❌ Вызов брошен не вам!", show_alert=True)
        return

    p1, p2 = get_pokedex(c_id)[0], get_pokedex(t_id)[0]
    p1_t, p2_t = POKEMON_TYPES.get(p1[0], "⚪️"), POKEMON_TYPES.get(p2[0], "⚪️")
    p1_m = 1.3 if p2_t in TYPE_ADVANTAGE.get(p1_t, []) else 1.0
    p2_m = 1.3 if p1_t in TYPE_ADVANTAGE.get(p2_t, []) else 1.0

    p1_cp, p2_cp = int(p1[4] * p1_m), int(p2[4] * p2_m)
    winner_id, loser_id = (c_id, t_id) if p1_cp >= p2_cp else (t_id, c_id)
    winner_p = p1[1] if p1_cp >= p2_cp else p2[1]

    change_user_coins(winner_id, DUEL_BET)
    change_user_coins(loser_id, -DUEL_BET)

    await callback.message.edit_text(
        f"💥 **ДУЭЛЬ ЗАВЕРШИЛАСЬ!**\n\n"
        f"🔴 **Боец 1:** {p1[1]} ({p1_t}) — **{p1_cp} CP**\n"
        f"🔵 **Боец 2:** {p2[1]} ({p2_t}) — **{p2_cp} CP**\n\n"
        f"🏆 Победил: **{winner_p}** (+{DUEL_BET} 🪙)!",
        parse_mode="Markdown"
    )
    await callback.answer()
