def generate_cp(rarity: str, is_shiny: bool, is_shadow: bool = False) -> int:
    min_cp, max_cp = CP_RANGES[rarity]
    cp = random.randint(min_cp, max_cp)
    if is_shiny: cp = int(cp * 1.15)
    if is_shadow: cp = int(cp * 1.20)
    return cp

def get_db():
    return sqlite3.connect("pokemon_bot.db")

def init_db():
    conn = get_db()
    cur = conn.cursor()
    cur.execute("""
        CREATE TABLE IF NOT EXISTS users (
            user_id INTEGER PRIMARY KEY,
            username TEXT,
            coins INTEGER DEFAULT 50,
            candies INTEGER DEFAULT 2,
            last_hunt REAL DEFAULT 0,
            lures INTEGER DEFAULT 0,
            ultraballs INTEGER DEFAULT 0,
            potions INTEGER DEFAULT 2,
            last_daily REAL DEFAULT 0,
            daily_streak INTEGER DEFAULT 0,
            egg_rarity TEXT DEFAULT NULL,
            egg_hatch_time REAL DEFAULT 0,
            referred_by INTEGER DEFAULT 0,
            vip_until REAL DEFAULT 0,
            expedition_tier INTEGER DEFAULT 0,
            expedition_end REAL DEFAULT 0,
            quiz_attempts INTEGER DEFAULT 3,
            quiz_reset_time REAL DEFAULT 0,
            last_spin REAL DEFAULT 0,
            breeding_end REAL DEFAULT 0,
            breeding_ready INTEGER DEFAULT 0
        );
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS inventory (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER,
            pokemon_id INTEGER,
            pokemon_name TEXT,
            rarity TEXT,
            is_shiny INTEGER DEFAULT 0,
            is_shadow INTEGER DEFAULT 0,
            is_purified INTEGER DEFAULT 0,
            cp INTEGER DEFAULT 100,
            UNIQUE(user_id, pokemon_name, is_shiny, is_shadow)
        );
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS towers (
            tower_id INTEGER PRIMARY KEY,
            name TEXT,
            req_cp INTEGER,
            champ_id INTEGER DEFAULT 0,
            champ_name TEXT DEFAULT 'Страж NPC',
            p_name TEXT,
            p_id INTEGER,
            p_cp INTEGER,
            rate_hr INTEGER DEFAULT 10,
            last_payout REAL DEFAULT 0
        );
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS market (
            lot_id INTEGER PRIMARY KEY AUTOINCREMENT,
            seller_id INTEGER,
            seller_name TEXT,
            pokemon_id INTEGER,
            pokemon_name TEXT,
            rarity TEXT,
            is_shiny INTEGER,
            is_shadow INTEGER,
            cp INTEGER,
            price INTEGER
        );
    """)
    conn.commit()

    # Миграция колонок для существующих баз
    cols = [
        ("potions", "INTEGER DEFAULT 2"), ("daily_streak", "INTEGER DEFAULT 0"),
        ("expedition_tier", "INTEGER DEFAULT 0"), ("expedition_end", "REAL DEFAULT 0"),
        ("quiz_attempts", "INTEGER DEFAULT 3"), ("quiz_reset_time", "REAL DEFAULT 0"),
        ("last_spin", "REAL DEFAULT 0"), ("breeding_end", "REAL DEFAULT 0"),
        ("breeding_ready", "INTEGER DEFAULT 0")
    ]
    for c_name, c_type in cols:
        try:
            cur.execute(f"ALTER TABLE users ADD COLUMN {c_name} {c_type}")
            conn.commit()
        except sqlite3.OperationalError:
            pass

    # Создание 3 башен
    towers_init = [
        (1, "🥉 Бронзовая Башня", 500, "Геодуд", 74, 550, 10),
        (2, "🥈 Серебряная Башня", 1500, "Мачамп", 68, 1650, 25),
        (3, "🥇 Золотая Башня", 2800, "Драгонайт", 149, 3100, 50)
    ]
    for tid, tname, req_cp, pname, pid, pcp, rate in towers_init:
        cur.execute("""
            INSERT OR IGNORE INTO towers (tower_id, name, req_cp, champ_id, champ_name, p_name, p_id, p_cp, rate_hr, last_payout)
            VALUES (?, ?, ?, 0, 'Страж Лиги', ?, ?, ?, ?, ?)
        """, (tid, tname, req_cp, pname, pid, pcp, rate, time.time()))
    conn.commit()
    cur.close()
    conn.close()

def get_user_data(user_id: int, username: str = ""):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT * FROM users WHERE user_id = ?", (user_id,))
    u = cur.fetchone()
    if not u:
        cur.execute("""
            INSERT INTO users (user_id, username, coins, candies, potions, quiz_attempts)
            VALUES (?, ?, 50, 2, 2, 3)
        """, (user_id, username))
        conn.commit()
        cur.close()
        conn.close()
        return get_user_data(user_id, username)

    col_names = [desc[0] for desc in cur.description]
    user_dict = dict(zip(col_names, u))
    if username and user_dict.get("username") != username:
        cur.execute("UPDATE users SET username = ? WHERE user_id = ?", (username, user_id))
        conn.commit()
        user_dict["username"] = username
    cur.close()
    conn.close()
    return user_dict

def update_user(user_id: int, **kwargs):
    conn = get_db()
    cur = conn.cursor()
    for k, v in kwargs.items():
        cur.execute(f"UPDATE users SET {k} = ? WHERE user_id = ?", (v, user_id))
    conn.commit()
    cur.close()
    conn.close()

def change_coins(user_id: int, delta: int):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("UPDATE users SET coins = MAX(0, COALESCE(coins, 0) + ?) WHERE user_id = ?", (delta, user_id))
    conn.commit()
    cur.close()
    conn.close()

def change_candies(user_id: int, delta: int):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("UPDATE users SET candies = MAX(0, COALESCE(candies, 0) + ?) WHERE user_id = ?", (delta, user_id))
    conn.commit()
    cur.close()
    conn.close()

def save_pokemon(user_id: int, pid: int, name: str, rarity: str, shiny: int, shadow: int, purified: int, cp: int) -> bool:
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT id FROM inventory WHERE user_id = ? AND pokemon_name = ? AND is_shiny = ? AND is_shadow = ?", (user_id, name, shiny, shadow))
    if cur.fetchone():
        cur.close()
        conn.close()
        return False
    cur.execute("""
        INSERT INTO inventory (user_id, pokemon_id, pokemon_name, rarity, is_shiny, is_shadow, is_purified, cp)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    """, (user_id, pid, name, rarity, shiny, shadow, purified, cp))
    conn.commit()
    cur.close()
    conn.close()
    return True

def get_inventory(user_id: int):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT id, pokemon_id, pokemon_name, rarity, is_shiny, is_shadow, is_purified, cp FROM inventory WHERE user_id = ? ORDER BY cp DESC", (user_id,))
    rows = cur.fetchall()
    cur.close()
    conn.close()
    return rows

ACTIVE_BATTLES = {}

def build_hp_bar(current_hp: int, max_hp: int) -> str:
    pct = max(0, min(1.0, current_hp / max_hp))
    filled = int(round(pct * 8))
    empty = 8 - filled
    return "🟩" * filled + "⬜️" * empty

def create_battle(p1_data: dict, p2_data: dict, battle_type: str, metadata: dict = None) -> str:
    battle_id = f"bt_{int(time.time() * 1000)}_{random.randint(100, 999)}"
    p1_max_hp = max(100, int(p1_data["cp"] * 0.45))
    p2_max_hp = max(100, int(p2_data["cp"] * 0.45))

    ACTIVE_BATTLES[battle_id] = {
        "id": battle_id,
        "type": battle_type,
        "metadata": metadata or {},
        "round": 1,
        "log": "Бой начинается! Выберите действие ниже:",
        "p1": {
            "id": p1_data["id"],
            "name": p1_data["name"],
            "poke_name": p1_data["poke_name"],
            "poke_id": p1_data["poke_id"],
            "type": POKEMON_TYPES.get(p1_data["poke_id"], "⚪️"),
            "cp": p1_data["cp"],
            "hp": p1_max_hp,
            "max_hp": p1_max_hp,
            "sp": 1,
            "shield": False,
            "potions": 1
        },
        "p2": {
            "id": p2_data["id"],
            "name": p2_data["name"],
            "poke_name": p2_data["poke_name"],
            "poke_id": p2_data["poke_id"],
            "type": POKEMON_TYPES.get(p2_data["poke_id"], "⚪️"),
            "cp": p2_data["cp"],
            "hp": p2_max_hp,
            "max_hp": p2_max_hp,
            "sp": 1,
            "shield": False,
            "potions": 1
        }
    }
    return battle_id

    p2 = b["p2"]
    p1_bar = build_hp_bar(p1["hp"], p1["max_hp"])
    p2_bar = build_hp_bar(p2["hp"], p2["max_hp"])
    sp1 = "🔵" * p1["sp"] + "⚪️" * (3 - p1["sp"])
    sp2 = "🔵" * p2["sp"] + "⚪️" * (3 - p2["sp"])

    text = (
        f"⚔️ **ПОШАГОВЫЙ БОЙ (Раунд {b['round']})**\n\n"

@dp.callback_query(F.data.startswith("b_act_"))
async def battle_action_handler(callback: CallbackQuery):
    _, _, b_id, action = callback.data.split("_")
    b = ACTIVE_BATTLES.get(b_id)
    if not b:
        await callback.answer("Бой завершен!", show_alert=True)
        return

    attacker = b["p1"] if callback.from_user.id == b["p1"]["id"] else b["p2"]
    defender = b["p2"] if callback.from_user.id == b["p1"]["id"] else b["p1"]
    
    if callback.from_user.id != attacker["id"]:
        await callback.answer("Сейчас не ваш ход!", show_alert=True)
        return

    log_event = ""
    if action == "atk":
        dmg = random.randint(18, 28)
        if defender["shield"]: dmg = int(dmg * 0.25)
        defender["hp"] = max(0, defender["hp"] - dmg)
        attacker["sp"] = min(3, attacker["sp"] + 1)
        log_event = f"⚔️ {attacker['name']} нанес быстрый удар: -{dmg} HP!"
    elif action == "spc":
        if attacker["sp"] < 2:
            await callback.answer("Нужно 2 SP для супер-удара!", show_alert=True)
            return
        attacker["sp"] -= 2
        mult = 1.5 if defender["type"] in TYPE_ADVANTAGE.get(attacker["type"], []) else 1.0
        dmg = int(random.randint(40, 58) * mult)
        if defender["shield"]: dmg = int(dmg * 0.25)
        defender["hp"] = max(0, defender["hp"] - dmg)
        bonus_str = " (Стихийный бонус!)" if mult > 1.0 else ""
        log_event = f"💥 {attacker['name']} обрушил супер-удар{bonus_str}: -{dmg} HP!"
    elif action == "def":
        attacker["shield"] = True
        log_event = f"🛡 {attacker['name']} встал в блок!"
    elif action == "pot":
        if attacker["potions"] <= 0:
            await callback.answer("Зелья закончились!", show_alert=True)
            return
        attacker["potions"] -= 1
        heal = int(attacker["max_hp"] * 0.40)
        attacker["hp"] = min(attacker["max_hp"], attacker["hp"] + heal)
        log_event = f"🧪 {attacker['name']} выпил зелье (+{heal} HP)!"

    defender["shield"] = False

    if defender["hp"] <= 0:
        await finish_battle(callback, b, winner=attacker, loser=defender)
        return

    # Ответный ход врага
    if b["p2"]["id"] == 0 or b["type"] in ["rocket", "tower"]:
        bot = b["p2"]
        user_p = b["p1"]
        b_act = "spc" if bot["sp"] >= 2 and random.random() < 0.6 else "atk"
        if b_act == "spc":
            bot["sp"] -= 2
            b_dmg = int(random.randint(35, 50))
            if user_p["shield"]: b_dmg = int(b_dmg * 0.25)
            user_p["hp"] = max(0, user_p["hp"] - b_dmg)
            log_event += f"\n🤖 Враг ответил супер-ударом: -{b_dmg} HP!"
        else:
            b_dmg = random.randint(15, 25)
            if user_p["shield"]: b_dmg = int(b_dmg * 0.25)
            user_p["hp"] = max(0, user_p["hp"] - b_dmg)
            bot["sp"] = min(3, bot["sp"] + 1)
            log_event += f"\n🤖 Враг атаковал: -{b_dmg} HP!"
        
        user_p["shield"] = False
        if user_p["hp"] <= 0:
            await finish_battle(callback, b, winner=bot, loser=user_p)
            return

    b["round"] += 1
    b["log"] = log_event
    new_text, new_markup = render_battle_card(b)
    try:
        await callback.message.edit_text(new_text, reply_markup=new_markup, parse_mode="Markdown")
    except Exception:
        pass
    await callback.answer()

async def finish_battle(callback: CallbackQuery, b: dict, winner: dict, loser: dict):

    if b["type"] == "rocket":
        if winner["id"] == b["p1"]["id"]:
            meta = b["metadata"]
            save_pokemon(winner["id"], meta["reward_pid"], meta["reward_name"], "Rare", 0, 1, 0, meta["reward_cp"])
            change_coins(winner["id"], 35)

                f"🎉 **ПОБЕДА НАД КОМАНДОЙ R!**\n\n"
                f"Вы спасли: **☠️ Теневой {meta['reward_name']} ({meta['reward_cp']} CP)**!\n"
                f"💰 Награда: **+35 🪙 монет**!",
                parse_mode="Markdown"
            )
        else:
            await callback.message.edit_text("💀 **ПОРАЖЕНИЕ!** Бандиты одолели вас.", parse_mode="Markdown")
    elif b["type"] == "tower":

            cur = conn.cursor()

                UPDATE towers

                WHERE tower_id = ?
            """, (winner["id"], winner["name"], winner["poke_name"], winner["poke_id"], winner["cp"], time.time(), tid))

@dp.callback_query(F.data.startswith("buy_lot_"))
async def buy_market_lot(callback: CallbackQuery):
    lid = int(callback.data.split("_")[2])
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT lot_id, seller_id, pokemon_id, pokemon_name, rarity, is_shiny, is_shadow, cp, price FROM market WHERE lot_id = ?", (lid,))
    lot = cur.fetchone()
    if not lot:
        cur.close()
        conn.close()
        await callback.answer("Лот уже продан!", show_alert=True)
        return

    u = get_user_data(callback.from_user.id, callback.from_user.username or "")
    if u["coins"] < lot[8]:
        cur.close()
        conn.close()
        await callback.answer("Недостаточно монет!", show_alert=True)
        return

    change_coins(callback.from_user.id, -lot[8])
    change_coins(lot[1], lot[8])
    save_pokemon(callback.from_user.id, lot[2], lot[3], lot[4], lot[5], lot[6], 0, lot[7])
    cur.execute("DELETE FROM market WHERE lot_id = ?", (lid,))
    conn.commit()
    cur.close()
    conn.close()

    await callback.message.edit_text(f"🎉 **Вы успешно купили {lot[3]} за {lot[8]} 🪙!**")
    await callback.answer()
