def get_pokemon_image_url(pokemon_id: int, is_shiny: bool) -> str:

    return f"https://raw.githubusercontent.com/PokeAPI/sprites/master/sprites/pokemon/other/official-artwork/{shiny_path}{pokemon_id}.png"

def generate_cp(rarity: str, is_shiny: bool, is_shadow: bool = False) -> int:
    min_cp, max_cp = CP_RANGES[rarity]
    cp = random.randint(min_cp, max_cp)
    if is_shiny: cp = int(cp * 1.15)
    if is_shadow: cp = int(cp * 1.20)
    return cp

def get_db():
    conn = sqlite3.connect("pokemon_bot.db")
    conn.execute("PRAGMA journal_mode = WAL;")
    conn.execute("PRAGMA synchronous = NORMAL;")
    return conn

def init_db():
    conn = get_db()
    cur = conn.cursor()
    cur.execute("""
        CREATE TABLE IF NOT EXISTS users (
            user_id INTEGER PRIMARY KEY,
            username TEXT,
            coins INTEGER DEFAULT 50,
            candies INTEGER DEFAULT 2,
            last_hunt REAL DEFAULT 0,
            lures INTEGER DEFAULT 0,
            ultraballs INTEGER DEFAULT 0,
            potions INTEGER DEFAULT 2,
            last_daily REAL DEFAULT 0,
            daily_streak INTEGER DEFAULT 0,
            egg_rarity TEXT DEFAULT NULL,
            egg_hatch_time REAL DEFAULT 0,
            referred_by INTEGER DEFAULT 0,
            vip_until REAL DEFAULT 0,
            expedition_tier INTEGER DEFAULT 0,
            expedition_end REAL DEFAULT 0,
            quiz_attempts INTEGER DEFAULT 3,
            quiz_reset_time REAL DEFAULT 0,
            last_spin REAL DEFAULT 0,
            breeding_end REAL DEFAULT 0,
            breeding_ready INTEGER DEFAULT 0
        );
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS inventory (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER,
            pokemon_id INTEGER,
            pokemon_name TEXT,
            rarity TEXT,
            is_shiny INTEGER DEFAULT 0,
            is_shadow INTEGER DEFAULT 0,
            is_purified INTEGER DEFAULT 0,
            cp INTEGER DEFAULT 100,
            UNIQUE(user_id, pokemon_name, is_shiny, is_shadow)
        );
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS towers (
            tower_id INTEGER PRIMARY KEY,
            name TEXT,
            req_cp INTEGER,
            champ_id INTEGER DEFAULT 0,
            champ_name TEXT DEFAULT 'Страж NPC',
            p_name TEXT,
            p_id INTEGER,
            p_cp INTEGER,
            rate_hr INTEGER DEFAULT 10,
            last_payout REAL DEFAULT 0
        );
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS market (
            lot_id INTEGER PRIMARY KEY AUTOINCREMENT,
            seller_id INTEGER,
            seller_name TEXT,
            pokemon_id INTEGER,
            pokemon_name TEXT,
            rarity TEXT,
            is_shiny INTEGER,
            is_shadow INTEGER,
            cp INTEGER,
            price INTEGER
        );
    """)
    cur.execute("CREATE INDEX IF NOT EXISTS idx_inv_uid ON inventory(user_id);")
    cur.execute("CREATE INDEX IF NOT EXISTS idx_towers_id ON towers(tower_id);")
    conn.commit()

    cols = [
        ("potions", "INTEGER DEFAULT 2"), ("daily_streak", "INTEGER DEFAULT 0"),
        ("expedition_tier", "INTEGER DEFAULT 0"), ("expedition_end", "REAL DEFAULT 0"),
        ("quiz_attempts", "INTEGER DEFAULT 3"), ("quiz_reset_time", "REAL DEFAULT 0"),
        ("last_spin", "REAL DEFAULT 0"), ("breeding_end", "REAL DEFAULT 0"),
        ("breeding_ready", "INTEGER DEFAULT 0"), ("vip_until", "REAL DEFAULT 0")
    ]
    for c_name, c_type in cols:
        try:
            cur.execute(f"ALTER TABLE users ADD COLUMN {c_name} {c_type}")
            conn.commit()
        except sqlite3.OperationalError:
            pass

    towers_init = [
        (1, "🥉 Башня Канто", 500, "Геодуд", 74, 550, 10),
        (2, "🥈 Башня Джото", 1500, "Скизор", 212, 1650, 25),
        (3, "🥇 Башня Хоэнна", 2800, "Райкваза", 384, 3200, 50)
    ]
    for tid, tname, req_cp, pname, pid, pcp, rate in towers_init:
        cur.execute("""
            INSERT OR IGNORE INTO towers (tower_id, name, req_cp, champ_id, champ_name, p_name, p_id, p_cp, rate_hr, last_payout)
            VALUES (?, ?, ?, 0, 'Страж Лиги', ?, ?, ?, ?, ?)
        """, (tid, tname, req_cp, pname, pid, pcp, rate, time.time()))
    conn.commit()
    cur.close()
    conn.close()

def get_user_data(user_id: int, username: str = ""):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT * FROM users WHERE user_id = ?", (user_id,))
    u = cur.fetchone()
    if not u:
        cur.execute("""
            INSERT INTO users (user_id, username, coins, candies, potions, quiz_attempts)
            VALUES (?, ?, 50, 2, 2, 3)
        """, (user_id, username))
        conn.commit()
        cur.close()
        conn.close()
        return get_user_data(user_id, username)

    col_names = [desc[0] for desc in cur.description]
    user_dict = dict(zip(col_names, u))
    if username and user_dict.get("username") != username:
        cur.execute("UPDATE users SET username = ? WHERE user_id = ?", (username, user_id))
        conn.commit()
        user_dict["username"] = username
    cur.close()
    conn.close()
    return user_dict

def update_user(user_id: int, **kwargs):
    conn = get_db()
    cur = conn.cursor()
    for k, v in kwargs.items():
        cur.execute(f"UPDATE users SET {k} = ? WHERE user_id = ?", (v, user_id))
    conn.commit()
    cur.close()
    conn.close()

def change_coins(user_id: int, delta: int):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("UPDATE users SET coins = MAX(0, COALESCE(coins, 0) + ?) WHERE user_id = ?", (delta, user_id))
    conn.commit()
    cur.close()
    conn.close()

    cur.execute("UPDATE users SET candies = MAX(0, COALESCE(candies, 0) + ?) WHERE user_id = ?", (delta, user_id))
    conn.commit()
    cur.close()
    conn.close()

def save_pokemon(user_id: int, pid: int, name: str, rarity: str, shiny: int, shadow: int, purified: int, cp: int) -> bool:
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT id FROM inventory WHERE user_id = ? AND pokemon_name = ? AND is_shiny = ? AND is_shadow = ?", (user_id, name, shiny, shadow))
    if cur.fetchone():
        cur.close()
        conn.close()
        return False
    cur.execute("""
        INSERT INTO inventory (user_id, pokemon_id, pokemon_name, rarity, is_shiny, is_shadow, is_purified, cp)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    """, (user_id, pid, name, rarity, shiny, shadow, purified, cp))
    conn.commit()
    cur.close()
    conn.close()
    return True

def admin_force_save_pokemon(user_id: int, pid: int, name: str, rarity: str, shiny: int, shadow: int, purified: int, cp: int):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT id, cp FROM inventory WHERE user_id = ? AND pokemon_name = ? AND is_shiny = ? AND is_shadow = ?", (user_id, name, shiny, shadow))
    row = cur.fetchone()
    if row:
        new_cp = max(row[1], cp)
        cur.execute("UPDATE inventory SET cp = ? WHERE id = ?", (new_cp, row[0]))
    else:
        cur.execute("""
            INSERT INTO inventory (user_id, pokemon_id, pokemon_name, rarity, is_shiny, is_shadow, is_purified, cp)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?)
        """, (user_id, pid, name, rarity, shiny, shadow, purified, cp))
    conn.commit()
    cur.close()
    conn.close()

def get_inventory(user_id: int):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT id, pokemon_id, pokemon_name, rarity, is_shiny, is_shadow, is_purified, cp FROM inventory WHERE user_id = ? ORDER BY cp DESC", (user_id,))
    rows = cur.fetchall()
    cur.close()
    conn.close()
    return rows

ACTIVE_BATTLES = {}

def build_hp_bar(current_hp: int, max_hp: int) -> str:
    pct = max(0.0, min(1.0, current_hp / max_hp))
    filled = int(round(pct * 8))
    empty = 8 - filled
    return "🟩" * filled + "⬜️" * empty

@dp.callback_query(F.data.startswith("bact:"))
async def battle_action_handler(callback: CallbackQuery):
    await callback.answer()
    parts = callback.data.split(":")
    if len(parts) != 3:
        return
    _, b_id, action = parts
    b = ACTIVE_BATTLES.get(b_id)
    if not b:
        return

    attacker = b["p1"] if callback.from_user.id == b["p1"]["id"] else b["p2"]
    defender = b["p2"] if callback.from_user.id == b["p1"]["id"] else b["p1"]
    
    if b["type"] == "duel":
        if callback.from_user.id != b["turn_user_id"]:
            return
    else:
        if callback.from_user.id != b["p1"]["id"]:
            return

    log_event = ""
    if action == "atk":
        dmg = random.randint(18, 28)
        if defender["shield"]: dmg = int(dmg * 0.25)
        defender["hp"] = max(0, defender["hp"] - dmg)
        attacker["sp"] = min(3, attacker["sp"] + 1)
        log_event = f"⚔️ {attacker['name']} нанес быстрый удар: -{dmg} HP!"
    elif action == "spc":
        if attacker["sp"] < 2:
            return
        attacker["sp"] -= 2
        mult = 1.5 if defender["type"] in TYPE_ADVANTAGE.get(attacker["type"], []) else 1.0
        dmg = int(random.randint(40, 58) * mult)
        if defender["shield"]: dmg = int(dmg * 0.25)
        defender["hp"] = max(0, defender["hp"] - dmg)
        bonus_str = " (Стихийный бонус!)" if mult > 1.0 else ""
        log_event = f"💥 {attacker['name']} обрушил супер-удар{bonus_str}: -{dmg} HP!"
    elif action == "def":
        attacker["shield"] = True
        log_event = f"🛡 {attacker['name']} встал в блок!"
    elif action == "pot":
        if attacker["potions"] <= 0:
            return
        attacker["potions"] -= 1
        update_user(attacker["id"], potions=attacker["potions"])
        heal = int(attacker["max_hp"] * 0.40)
        attacker["hp"] = min(attacker["max_hp"], attacker["hp"] + heal)
        log_event = f"🧪 {attacker['name']} выпил зелье (+{heal} HP)!"

    defender["shield"] = False

    if defender["hp"] <= 0:
        await finish_battle(callback, b, winner=attacker, loser=defender)
        return

    if b["type"] == "duel":
        b["turn_user_id"] = defender["id"]
        if b["turn_user_id"] == b["p1"]["id"]:
            b["round"] += 1
    elif b["p2"]["id"] == 0 or b["type"] in ["rocket", "tower"]:
        bot = b["p2"]
        user_p = b["p1"]
        b_act = "spc" if bot["sp"] >= 2 and random.random() < 0.6 else "atk"
        if b_act == "spc":
            bot["sp"] -= 2
            b_dmg = int(random.randint(35, 50))
            if user_p["shield"]: b_dmg = int(b_dmg * 0.25)
            user_p["hp"] = max(0, user_p["hp"] - b_dmg)
            log_event += f"\n🤖 Враг ответил супер-ударом: -{b_dmg} HP!"
        else:
            b_dmg = random.randint(15, 25)
            if user_p["shield"]: b_dmg = int(b_dmg * 0.25)
            user_p["hp"] = max(0, user_p["hp"] - b_dmg)
            bot["sp"] = min(3, bot["sp"] + 1)
            log_event += f"\n🤖 Враг атаковал: -{b_dmg} HP!"
        
        user_p["shield"] = False
        if user_p["hp"] <= 0:
            await finish_battle(callback, b, winner=bot, loser=user_p)
            return
        b["round"] += 1

    b["log"] = log_event
    new_text, new_markup = render_battle_card(b)
    try:
        await callback.message.edit_text(new_text, reply_markup=new_markup, parse_mode="HTML")
    except Exception:
        pass

async def finish_battle(callback: CallbackQuery, b: dict, winner: dict, loser: dict):
    del ACTIVE_BATTLES[b["id"]]
    if b["type"] == "rocket":
        if winner["id"] == b["p1"]["id"]:
            meta = b["metadata"]
            save_pokemon(winner["id"], meta["reward_pid"], meta["reward_name"], "Rare", 0, 1, 0, meta["reward_cp"])
            change_coins(winner["id"], 35)
            await callback.message.edit_text(
                f"🎉 <b>ПОБЕДА НАД КОМАНДОЙ R!</b>\n\n"
                f"Вы спасли: <b>☠️ Теневой {html.escape(meta['reward_name'])} ({meta['reward_cp']} CP)</b>!\n"
                f"💰 Награда: <b>+35 🪙 монет</b>!",
                parse_mode="HTML"
            )
        else:
            await callback.message.edit_text("💀 <b>ПОРАЖЕНИЕ!</b> Бандиты одолели вас.", parse_mode="HTML")
    elif b["type"] == "tower":
        if winner["id"] == b["p1"]["id"]:
            tid = b["metadata"]["tower_id"]
            conn = get_db()
            cur = conn.cursor()
            cur.execute("SELECT champ_id, last_payout, rate_hr, name FROM towers WHERE tower_id = ?", (tid,))
            old_t = cur.fetchone()
            
            # Выплата дани старому чемпиону
            if old_t and old_t[0] > 0:
                hours = max(1, int((time.time() - old_t[1]) / 3600))
                payout = hours * old_t[2]
                change_coins(old_t[0], payout)
                try:
                    await callback.bot.send_message(
                        old_t[0],
                        f"🏰 Вашего бойца выбили из <b>{old_t[3]}</b>!\n💰 Накопленная дань: <b>+{payout} 🪙 монет</b> начислена на баланс.",
                        parse_mode="HTML"
                    )
                except Exception:
                    pass

            cur.execute("""
                UPDATE towers 
                SET champ_id = ?, champ_name = ?, p_name = ?, p_id = ?, p_cp = ?, last_payout = ?
                WHERE tower_id = ?
            """, (winner["id"], winner["name"], winner["poke_name"], winner["poke_id"], winner["cp"], time.time(), tid))
            conn.commit()
            cur.close()
            conn.close()

# ================= 6. 👑 АДМИН-КОМАНДЫ =================
def resolve_target_user(message: types.Message, args: list):
    if message.reply_to_message and message.reply_to_message.from_user:
        target = message.reply_to_message.from_user
        return target.id, target.first_name, args

    if not args:
        return message.from_user.id, message.from_user.first_name, []

    first_arg = args[0]
    if first_arg.isdigit() and len(first_arg) >= 6:
        uid = int(first_arg)
        conn = get_db()
        cur = conn.cursor()
        cur.execute("SELECT user_id, username FROM users WHERE user_id = ?", (uid,))
        row = cur.fetchone()
        cur.close()
        conn.close()
        name = f"@{row[1]}" if (row and row[1]) else f"ID {uid}"
        return uid, name, args[1:]

    if first_arg.startswith("@") or (len(args) >= 2 and not first_arg.isdigit() and not ALL_POKEMON_MAP.get(first_arg.lower())):
        clean = first_arg.lstrip("@").lower()
        conn = get_db()
        cur = conn.cursor()
        cur.execute("SELECT user_id, username FROM users WHERE LOWER(username) = ?", (clean,))
        row = cur.fetchone()
        cur.close()
        conn.close()
        if row:
            return row[0], f"@{row[1]}", args[1:]
        else:
            return None, f"@{clean}", args[1:]

    return message.from_user.id, message.from_user.first_name, args

    if not target_id:
        await message.answer(f"❌ Игрок {target_name} не найден в базе данных.", parse_mode="HTML")
        return

    if not p_args:
        await message.answer("❌ Укажите номер (1-386) или имя покемона.", parse_mode="HTML")
        return

    is_shiny = 1 if "shiny" in [a.lower() for a in p_args] else 0
    clean_p_words = [a for a in p_args if a.lower() != "shiny"]
    p_query = " ".join(clean_p_words).strip().lower()

    if p_query.isdigit():
        pid = int(p_query)
    else:
        pid = ALL_POKEMON_MAP.get(p_query, {}).get("id")

    if not pid or pid < 1 or pid > 386:
        await message.answer("❌ Покемон с таким номером/именем не найден (диапазон 1–386).", parse_mode="HTML")
        return

    pinfo = ALL_POKEMON_MAP.get(pid, {"name": "Пикачу", "rarity": "Rare"})
    cp = generate_cp(pinfo["rarity"], bool(is_shiny))
    admin_force_save_pokemon(target_id, pid, pinfo["name"], pinfo["rarity"], is_shiny, 0, 0, cp)

    shiny_tag = "✨ ШАЙНИ " if is_shiny else ""
    await message.answer(f"👑 <b>{shiny_tag}{html.escape(pinfo['name'])} #{pid:03d} ({cp} CP)</b> выдан игроку <b>{html.escape(target_name)}</b>!", parse_mode="HTML")
    if target_id != message.from_user.id:
        try:
            await message.bot.send_message(target_id, f"🎁 <b>Подарок от Администратора:</b> вам выдан {shiny_tag}<b>{html.escape(pinfo['name'])} #{pid:03d} ({cp} CP)</b>!", parse_mode="HTML")
        except Exception:
            pass

    if not raw_args:
        await message.answer("Использование: <code>/give_coins 500</code> или <code>/give_coins @username 500</code>", parse_mode="HTML")
        return
    
    target_id, target_name, c_args = resolve_target_user(message, raw_args)
    if not target_id or not c_args or not c_args[0].isdigit():
        await message.answer("❌ Укажите количество монет числом.", parse_mode="HTML")
        return
    
    count = int(c_args[0])
    change_coins(target_id, count)
    await message.answer(f"👑 Начислено <b>+{count} 🪙 монет</b> игроку <b>{html.escape(target_name)}</b>!", parse_mode="HTML")

@dp.message(Command("give_items"))
async def admin_give_items(message: types.Message, command: CommandObject):

    update_user(message.from_user.id, lures=u["lures"] + count, ultraballs=u["ultraballs"] + count, candies=u["candies"] + count, potions=u.get("potions", 0) + count)
    await message.answer(f"👑 Начислено по <b>+{count}</b> приманок, ультраболов, зелий и конфет!", parse_mode="HTML")
